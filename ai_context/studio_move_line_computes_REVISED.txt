=== x_studio_affected_2nd_uom  [GROUP A] ===
==============================================================================
for record in self:
    if record.picking_id and (record['picking_code'] == 'outgoing' or not record['picking_code']):
        location = record['location_id']

        for quant in location['quant_ids']:
            if (quant['product_id'] == record['product_id']
                    and quant['owner_id'] == record['owner_id']
                    and quant['lot_id'] == record['lot_id']):

                if quant['reservation_tags']:
                    picked_move_lines = self.env['stock.move.line'].search([
                        ('picking_id', 'in', quant['reservation_tags'].ids),
                        ('lot_id', '=', quant['lot_id'].id),
                    ])

                    total_picked_2nd_uom = sum(picked_move_lines.mapped('x_studio_affected_2nd_uom'))

                    record['x_studio_affected_2nd_uom'] = quant['x_studio_2nd_uom'] - total_picked_2nd_uom
                else:
                    record['x_studio_affected_2nd_uom'] = quant['x_studio_2nd_uom']

                break

    else:
        record['x_studio_affected_2nd_uom'] = 0
